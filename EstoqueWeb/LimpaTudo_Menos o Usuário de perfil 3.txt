-- Limpa Entradas e Estoque de todas as empresas

START TRANSACTION;

-- 1. Apaga todas as movimentações de entrada
-- de todas as empresas
DELETE FROM movimentacoes;


-- 2. Apaga todos os itens das entradas
DELETE FROM itens_entrada;


-- 3. Apaga todas as entradas de estoque
DELETE FROM entradas_estoque;

-- 4. Apaga os produtos
DELETE from produtos;

--categorias
DELETE from categorias;

-- usuarios
DELETE from usuarios where perfil_id <> 3;

-- 5. Confere a limpeza
SELECT
    COUNT(*) AS total_produtos,
    SUM(estoque_atual) AS estoque_total
FROM produtos;

SELECT COUNT(*) AS total_entradas
FROM entradas_estoque;

SELECT COUNT(*) AS total_itens_entrada
FROM itens_entrada;

SELECT COUNT(*) AS total_movimentacoes_entrada
FROM movimentacoes
WHERE tipo = 'ENTRADA';


-- CONFIRMA A OPERAÇÃO
COMMIT;
